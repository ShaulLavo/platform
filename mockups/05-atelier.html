import { createThread } from "@/features/thread/service";
import { runPlan } from "@/features/plan/runner";
import { restoreCheckpoint } from "@/features/checkpoint";
import { createStructuredError } from "@/lib/structured-errors";
type OrchestratorOpts = {
workspace: string;
model: ModelId;
maxTurns?: number;
};
// Drive plan→act turns until the plan reports done.
export async function orchestrate(opts: OrchestratorOpts) {
const thread = await createThread(opts.workspace);
await restoreCheckpoint(thread, "latest");
for (let turn = 0; turn < (opts.maxTurns ?? 12); turn++) {
const step = await runPlan(thread, opts.model);
if (step.done) return thread.commit();
if (step.aborted) break;
}
throw createStructuredError("ORCHESTRATOR_EXHAUSTED");
}
